import { describe, expect, test, jest, beforeEach } from '@jest/globals';
import { detectRisk, runRiskDetection } from '../utils/cron';
import prisma from '../utils/prismaClient';
import { createNotification } from '../utils/notification.utils';
import logger from '../utils/logger';
 
jest.mock('../utils/prismaClient', () => ({
  __esModule: true,
  default: {
    student: {
      findMany: jest.fn(),
    },
    studentSuccessMetric: {
      upsert: jest.fn(),
    },
    user: {
      findFirst: jest.fn(),
    },
  },
}));
 
jest.mock('../utils/notification.utils', () => ({
  createNotification: jest.fn(),
}));
 
jest.mock('../utils/logger', () => ({
  info: jest.fn(),
  error: jest.fn(),
  warn: jest.fn(),
}));
 
describe('Academic Risk Detection', () => {
  beforeEach(() => {
    jest.clearAllMocks();
  });
 
  describe('detectRisk function', () => {
    // Case 1: GPA below threshold (< 2.0) → isAtRisk: true, reason: 'low_gpa'
    test('should detect risk when GPA is below 2.0', () => {
      const studentData = {
        gpa: 1.8,
        absenceRate: 10,
        incompleteGrades: 0,
      };
      const result = detectRisk(studentData);
      expect(result.isAtRisk).toBe(true);
      expect(result.reason).toBe('low_gpa');
    });
 
    // Case 2: Absence rate above 25% → isAtRisk: true, reason: 'high_absences'
    test('should detect risk when absence rate is above 25%', () => {
      const studentData = {
        gpa: 3.0,
        absenceRate: 30,
        incompleteGrades: 0,
      };
      const result = detectRisk(studentData);
      expect(result.isAtRisk).toBe(true);
      expect(result.reason).toBe('high_absences');
    });
 
    // Case 3: More than 2 incomplete grades → isAtRisk: true, reason: 'incomplete_grades'
    test('should detect risk when there are more than 2 incomplete grades', () => {
      const studentData = {
        gpa: 3.0,
        absenceRate: 10,
        incompleteGrades: 3,
      };
      const result = detectRisk(studentData);
      expect(result.isAtRisk).toBe(true);
      expect(result.reason).toBe('incomplete_grades');
    });
 
    // Case 4: Healthy student (GPA 3.5, absences 10%, no incompletes) → isAtRisk: false
    test('should not detect risk for a healthy student', () => {
      const studentData = {
        gpa: 3.5,
        absenceRate: 10,
        incompleteGrades: 0,
      };
      const result = detectRisk(studentData);
      expect(result.isAtRisk).toBe(false);
      expect(result.reason).toBeUndefined();
    });
  });
 
  describe('runRiskDetection cron logic', () => {
    const mockStudent = {
      id: 1,
      userId: 1,
      firstName: 'John',
      lastName: 'Doe',
      studentId: 'STU001',
      departmentId: 1,
      attendance: [{ status: 'ABSENT' }, { status: 'ABSENT' }, { status: 'ABSENT' }, { status: 'ABSENT' }], // 100% absence
      quizSubmissions: [{ score: 30 }], // 30/100 -> GPA 1.2
      taskSubmissions: [],
      courses: [
        { tasks: [{ id: 1 }, { id: 2 }, { id: 3 }] }
      ]
    };
 
    // Case 5: Risk detected → notification created
    test('should create notification when risk is detected', async () => {
      (prisma.student.findMany as any).mockResolvedValueOnce([mockStudent]).mockResolvedValueOnce([]);
      (prisma.user.findFirst as any).mockResolvedValue({ id: 2, role: 'DEPARTMENT_ADMIN' });
      (prisma.studentSuccessMetric.upsert as any).mockResolvedValue({});
 
      await runRiskDetection();
 
      expect(createNotification).toHaveBeenCalled();
      expect(createNotification).toHaveBeenCalledWith(expect.objectContaining({
        userId: mockStudent.userId,
        title: 'Academic Alert'
      }));
    });
 
    // Case 6: No risk → NO notification created
    test('should not create notification when no risk is detected', async () => {
      const healthyStudent = {
        ...mockStudent,
        attendance: [{ status: 'PRESENT' }, { status: 'PRESENT' }, { status: 'PRESENT' }, { status: 'PRESENT' }],
        quizSubmissions: [{ score: 90 }],
        taskSubmissions: [{ taskId: 1 }, { taskId: 2 }, { taskId: 3 }]
      };
      (prisma.student.findMany as any).mockResolvedValueOnce([healthyStudent]).mockResolvedValueOnce([]);
      (prisma.studentSuccessMetric.upsert as any).mockResolvedValue({});
 
      await runRiskDetection();
 
      expect(createNotification).not.toHaveBeenCalled();
    });
 
    // Case 7: DB error during cron → error logged, does NOT crash the process
    test('should log error and not crash on DB failure', async () => {
      (prisma.student.findMany as any).mockRejectedValue(new Error('DB Error'));
 
      await expect(runRiskDetection()).resolves.not.toThrow();
      expect(logger.error).toHaveBeenCalledWith(expect.stringContaining('AI Risk Detection error'), expect.any(Object));
    });
  });
});
